'🔥 Promoção ATIVA' : 'Promoção desativada'; if (label) label.style.color = ativa ? '#ff6600' : 'var(--muted)'; } function togglePromocaoTodos(todos) { const div = document.getElementById('promo-cats'); if (div) div.style.display = todos ? 'none' : 'flex'; } function mudarTipoValidade(tipo) { document.getElementById('promo-horario').style.display = tipo === 'horario' ? 'block' : 'none'; document.getElementById('promo-data').style.display = tipo === 'data' ? 'block' : 'none'; } async function salvarPromocao() { const cats = [...document.querySelectorAll('#promo-cats input:checked')].map(i => i.value); const todos = document.getElementById('promo-todos').checked; const tipo = document.getElementById('promo-tipo').value; if (!configData) configData = {}; configData.promocao = { ativa: document.getElementById('promo-ativa').checked, titulo: document.getElementById('promo-titulo').value.trim(), desconto: parseInt(document.getElementById('promo-desconto').value) || 15, todos, categorias: todos ? [] : cats, validade: { tipo, hora_inicio: document.getElementById('promo-hora-ini')?.value || '', hora_fim: document.getElementById('promo-hora-fim')?.value || '', data_inicio: document.getElementById('promo-data-ini')?.value || '', data_fim: document.getElementById('promo-data-fim')?.value || '', } }; setStatus('Salvando promoção...', 'info'); try { const atual = await ghGet('data/config.json'); shaConfig = atual.sha; const res = await ghPut('data/config.json', JSON.stringify(configData, null, 4), shaConfig, 'Admin: promoção do dia'); shaConfig = res.content.sha; setStatus('✅ Promoção salva! Aguarde ~1 min para atualizar.', 'ok'); setTimeout(() => setStatus('', ''), 6000); } catch(e) { setStatus('Erro: ' + e.message, 'erro'); } } async function salvarComboDesconto() { if (!configData) configData = {}; configData.combo = { desconto: parseInt(document.getElementById('combo-desconto').value) || 10 }; setStatus('Salvando desconto do combo...', 'info'); try { const atual = await ghGet('data/config.json'); shaConfig = atual.sha; const res = await ghPut('data/config.json', JSON.stringify(configData, null, 4), shaConfig, 'Admin: desconto combo'); shaConfig = res.content.sha; setStatus('✅ Desconto do combo salvo!', 'ok'); setTimeout(() => setStatus('', ''), 5000); } catch(e) { setStatus('Erro: ' + e.message, 'erro'); } } // ============================ // FIM DO PAINEL ADMIN // ============================
